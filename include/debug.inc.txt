;Вывод на экран
print_char:
    push ax
    push bx
    mov ah, 0x0E
    int 0x10
    pop bx
    pop ax

    ret

print_string:
    push ax
    push si

.loop:
    lodsb
    test al, al
    jz .done
    call print_char
    jmp .loop

.done:
    pop si
    pop ax

    ret

;Отладка через порт 0xE9 (арифметический метод)
print_debug:
    push ax
    push si
    push dx
    mov dx, 0xE9

.loop:
    lodsb
    test al, al
    jz .done
    out dx, al
    jmp .loop

.done:
    pop dx
    pop si
    pop ax

    ret

; Вывод байта в hex (AL) через порт 0xE9 (сохраняет все регистры)
debug_hex_byte:
    push ax
    push bx
    push dx
    mov bl, al
    shr al, 4
    call debug_hex_nibble
    mov al, bl
    and al, 0x0F
    call debug_hex_nibble
    pop dx
    pop bx
    pop ax

    ret

debug_hex_nibble:
    ;push dx
    add al, '0'
    cmp al, '9'
    jle .print
    add al, 7

.print:
    mov dx, 0xE9
    out dx, al
    ;pop dx

    ret

;Вывод слова в hex (AX) через порт 0xE9 (сохраняет все регистры)
debug_hex_word:
    push ax
    xchg al, ah
    call debug_hex_byte
    pop ax
    call debug_hex_byte

    ret

debug_out_char:
    push ax
    push dx
    mov dx, 0xE9
    out dx, al
    pop dx
    pop ax

    ret


dump_of_catalogue:
	;ДАМП КОРНЕВОГО КАТАЛОГА (первые 32 байта)
	push si
	push es
	push ds
    mov dx, 0xE9
    mov si, msg_root_dump
    call print_debug
    push es
    push ds
    mov ax, 0x8000
    mov ds, ax
    mov si, 0
    mov cx, 32

dump_root:
    lodsb
    call debug_hex_byte
    mov al, ' '
    out dx, al
    loop dump_root
    mov al, 13
    out dx, al
    mov al, 10
    out dx, al
    pop ds
    pop es
    xor ax, ax
    mov ds, ax
	pop ds
	pop es
	pop si

	ret

debug_hex_dword:
    push eax
    push ebx
    push ecx
    mov ebx,eax

;Вывод символа AL на экран (BIOS Teletype)
screen_print_char:
    push ax
    push bx
    mov ah, 0x0E
    mov bx, 0x0007  ; Страница 0, цвет белый
    int 0x10
    pop bx
    pop ax
    ret

;Вывод строки SI на экран
screen_print_string:
    push ax
    push si
.s_loop:
    lodsb
    test al, al
    jz .s_done
    call screen_print_char
    jmp .s_loop
.s_done:
    pop si
    pop ax
    ret

;Вывод ниббла (4 бита) AL на экран в HEX
screen_hex_nibble:
    push ax
    and al, 0x0F
    add al, '0'
    cmp al, '9'
    jle .sh_print
    add al, 7         ; 'A'-'F'
.sh_print:
    call screen_print_char
    pop ax
    ret

;Вывод байта AL на экран в HEX
screen_hex_byte:
    push ax
    mov bl, al        ;Сохраняем байт в BL

    shr al, 4         ;Старший ниббл
    call screen_hex_nibble

    mov al, bl        ;Младший ниббл
    call screen_hex_nibble

    pop ax
    ret

;Вывод слова AX на экран в HEX
screen_hex_word:
    push ax
    xchg al, ah       ; Старший байт первый
    call screen_hex_byte
    pop ax
    call screen_hex_byte
    ret

;Вывод dword EAX на экран в HEX
screen_hex_dword:
    push eax
    push ebx
    push ecx
    mov ebx, eax
    mov ecx, 8        ;8 нибблов
.shd_loop:
    rol ebx, 4
    mov al, bl
    call screen_hex_nibble
    loop .shd_loop
    pop ecx
    pop ebx
    pop eax
    ret

;Вывод переноса строки на экран
screen_newline:
    push ax
    mov al, 13
    call screen_print_char
    mov al, 10
    call screen_print_char
    pop ax
    ret
