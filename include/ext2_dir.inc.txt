;Парсер директорий EXT2
BITS 16
DIR_SEG equ 0x1600

ext2_find_file:
    push bx
    push cx
    push dx
    push di
    push si
    push ds
    push es

    ;Сохраняем указатели на имя искомого файла
    mov [cs:search_name_off], si
    mov [cs:search_name_seg], ds

    ;Вычисляем длину строки искомого файла
    push si
    xor cx, cx
.calc_name_len:
    cmp byte [si], 0
    je .name_len_done
    inc cx
    inc si
    jmp .calc_name_len
.name_len_done:
    mov [cs:search_name_len], cx
    pop si

    ;Читаем инвод корневой директории (Inode №2)
    mov ax, 2
    call ext2_inode_to_group
    call ext2_get_group_inode_table
    mov ax, 2
    call ext2_read_inode
    jc .inode_err

    ;Загружаем все блоки данных корневой директории в память DIR_SEG
    mov si, [cs:inode_offset]
    call ext2_read_root_inode_blocks
    jc .dir_err

    ;Начинаем обход загруженных блоков директории
    mov ax, 0
    mov [cs:dir_current_block], ax

.next_dir_block:
    mov dx, [cs:dir_current_block]
    cmp dx, [cs:dir_block_count]
    jae .not_found          ;Если обошли все блоки и не нашли — ошибка

    ;Рассчитываем сегмент текущего блока: каждый блок (1024 байта) занимает 0x40 параграфов
    mov ax, dx
    mov bx, 0x0040          ;1024 байта = 0x40 параграфов памяти
    mul bx
    add ax, DIR_SEG         ;Смещение сегмента для текущего блока
    mov ds, ax              ;DS указывает строго на начало текущего блока директории

    xor di, di              ;DI = смещение внутри текущего блока (0)

.search_loop:
    ;Проверка на выход за границы блока (размер блока строго 1024 байта)
    cmp di, 1024
    jae .block_done

    ;Читаем 32-битный номер инвода текущей записи
    mov ax, [di]
    mov dx, [di+2]
    or ax, dx
    jz .skip_empty_entry    ;Если инвод равен 0 — запись удалена, пропускаем

    ; Запись валидна. Читаем rec_len (смещение 4) и name_len (смещение 6)
    mov bx, [di+4]          ;BX = rec_len (на сколько байт сдвинуть DI для следующей записи)
    movzx cx, byte [di+6]   ;CX = name_len (берем строго 1 байт для совместимости с rev1)

    ;Проверяем, совпадает ли длина имени текущей записи с искомой
    cmp cx, [cs:search_name_len]
    jne .advance_di         ;Если длины разные, даже не сравниваем строки

    ; Подготавливаем регистры для побайтового сравнения строк
    push cx
    push si
    push di
    push ds

    add di, 8               ;DI + 8 = Начало строки имени внутри записи директории
    mov ax, [cs:search_name_seg]
    mov es, ax
    mov bx, [cs:search_name_off] ;ES:BX указывает на строку с именем файла

.compare_strings:
    mov al, [di]            ;Байт из ФС
    mov ah, [es:bx]         ;Байт из загрузчика
    cmp al, ah
    jne .strings_differ
    inc di
    inc bx
    loop .compare_strings

    ;Строки совпали, файл найден
    pop ds
    pop di
    pop si
    pop cx

    mov ax, [di]            ;Извлекаем младшее слово номера найденного инвода ядра
    clc                     ;Сбрасываем флаг переноса (Успех)
    jmp .exit

.strings_differ:
    pop ds
    pop di
    pop si
    pop cx

.advance_di:
    ;Безопасный шаг к следующей записи ФС
    mov bx, [di+4]          ;Снова берем rec_len текущей записи
    test bx, bx
    jz .block_done          ;Защита - если rec_len равен 0, блок поврежден, выходим
    add di, bx              ;Сдвигаем DI вперед
    jmp .search_loop

.skip_empty_entry:
    mov bx, [di+4]          ;Для пустой записи тоже берем rec_len
    test bx, bx
    jz .block_done
    add di, bx
    jmp .search_loop

.block_done:
    ;Переходим к парсингу следующего загруженного блока директории
    inc word [cs:dir_current_block]
    jmp .next_dir_block

.not_found:
    stc                     ;Выставляем флаг переноса (Файл не найден)
    jmp .exit
.inode_err:
.dir_err:
    stc

.exit:
    pop es
    pop ds
    pop si
    pop di
    pop dx
    pop cx
    pop bx
    ret

ext2_read_root_inode_blocks:
    push ax
    push bx
    push cx
    push dx
    push si
    push di
    push ds
    push es

    mov ax,INODE_SEG
    mov ds,ax

    add si,40
    xor di,di

.next_block:
    cmp di,12
    jae .done
    mov bx,di
    shl bx,1
    shl bx,1
    mov ax,[si+bx]
    test ax,ax
    jz .done

    push ax
    mov ax,di
    call mul1024
    mov bx,ax
    mov ax,DIR_SEG
    mov es,ax
    pop ax

    cmp word [cs:ext2_block_size],1024
    jne .error
    shl ax,1
    xor dx,dx
    mov cx,2
    call ext2_read_sectors
    jc .error
    inc di
    jmp .next_block

.done:
    mov [cs:dir_block_count],di
    pop es
    pop ds
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    clc
    ret

.error:
    pop es
    pop ds
    pop di
    pop si
    pop dx
    pop cx
    pop bx
    pop ax
    stc
    ret

mul1024:
    push cx
    mov cl,10
    shl ax,cl
    pop cx
    ret


;Variables

search_name_off   dw 0
search_name_seg   dw 0
dir_block_count   dw 0
dir_current_block dw 0
search_name_len   dw 0
dir_entry_rec_len dw 0
