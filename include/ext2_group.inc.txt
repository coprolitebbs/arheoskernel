;Загрузчик таблиц групп
BITS 16
GDT_SEG equ 0x1200

ext2_read_group_table:
    ;Сохранение абсолютно всех регистров процессора
    pusha
    push ds
    push es

    ;Загружаем s_first_data_block из Суперблока (сегмент SUPER_SEG = 0x1000)
    mov ax, 0x1000
    mov ds, ax
    mov ax, [20]       ;AX = s_first_data_block (обычно 1 для блока 1КБ)
    inc ax             ;AX = Номер логического блока таблицы групп (1 + 1 = 2)

    ;Переводим логический блок EXT2 в относительный сектор ФС
    ;Так как размер блока 1024 байта, а сектор 512: умножаем на 2 (shl ax, 1)
    shl ax, 1
    xor dx, dx         ;DX:AX = Младший относительный сектор таблицы групп (4)

    ;Настраиваем ES:BX на целевой буфер в памяти (0x1200:0000)
    mov bx, GDT_SEG
    mov es, bx
    xor bx, bx         ;Буфер готов: 0x1200:0000

    ;Читаем строго 2 сектора (1 логический блок EXT2 = 1024 байта).
    ;Этого с избытком хватает для дескрипторов групп на диске до 2 ГБ.
    ;Защищает от выхода за границы размеченной области диска.
    mov cx, 2
    call ext2_read_sectors
    jc .error

    ;Успешный выход -восстанавливаем контекст в первозданном виде
    pop es
    pop ds
    popa
    clc
    ret

.error:
    ;Выводим маркер ошибки таблицы групп в порт отладки QEMU
    mov al, '@'
    out 0xE9, al

    pop es
    pop ds
    popa
    stc
    ret


; Вспомогательные функции

ext2_get_group_inode_table:
    push ax
    push bx
    push dx
    push si
    push ds

    ;На входе AX = номер группы. Умножаем на 32 (размер дескриптора группы)
    xor dx, dx
    mov bx, 32
    mul bx
    mov si, ax

    mov ax, GDT_SEG
    mov ds, ax

    ;Смещение поля bg_inode_table внутри структуры ext2_group_desc — 8 байт
    mov bx, si
    add bx, 8

    mov ax, [bx]       ;Младшее слово номера блока
    mov dx, [bx+2]     ;Старшее слово номера блока

    pop ds
    pop si
    pop dx
    pop bx
    pop ax
    ret

ext2_inode_to_group:
    push bx
    dec ax             ;Индексы инводов начинаются с 1, переводим в 0-индекс
    xor dx, dx
    mov bx, 8192       ;s_inodes_per_group
    div bx             ;AX = номер группы, DX = относительный индекс инвода в группе
    pop bx
    ret
