;Низкоуровневые функции Stage 2

BITS 16

lba_supported db 0
heads_tmp     dw 16
sectors_tmp   dw 63
heads         db 0
sectors       db 0
read_lba_low  dw 0
read_lba_high dw 0
read_count    dw 0
read_buf_off  dw 0
read_buf_seg  dw 0
tmp_lba_low  dw 0
tmp_lba_high dw 0

EXT2_START_LBA equ 64


; Disk Address Packet

align 4
dap:
    db 16
    db 0
    dw 0
    dw 0
    dw 0
    dd 0
    dd 0
times 32 db 0


;ext2_disk_init & ext2_get_geometry

ext2_disk_init:
    mov ah,00h
    mov dl,[drive]
    int 13h
    mov ah,41h
    mov bx,55AAh
    mov dl,[drive]
    int 13h
    jc .chs
    cmp bx,0AA55h
    jne .chs
    mov byte [lba_supported],1
    ret
.chs:
    mov byte [lba_supported],0
    call ext2_get_geometry
    ret

ext2_get_geometry:
    mov ah,08h
    mov dl,[drive]
    int 13h
    jc .default
    mov [heads],dh
    inc byte [heads]
    mov al,cl
    and al,3Fh
    mov [sectors],al
    xor ax,ax
    mov al,[heads]
    mov [heads_tmp],ax
    xor ax,ax
    mov al,[sectors]
    mov [sectors_tmp],ax
    ret
.default:
    mov word [heads_tmp],16
    mov word [sectors_tmp],63
    ret


; ext2_read_sectors

ext2_read_sectors:
    add ax, EXT2_START_LBA
    adc dx, 0

    pushf
    push ds
    pusha

    mov bp, sp

    xor ax, ax
    mov ds, ax

    mov si, dap
    mov byte [si], 16
    mov byte [si+1], 0

    mov cx, [bp+12]
    mov bx, [bp+8]

    mov [si+2], cx
    mov [si+4], bx
    mov [si+6], es

    mov ax, [bp+14]
    mov dx, [bp+10]

    mov [si+8], ax
    mov [si+10], dx
    mov word [si+12], 0
    mov word [si+14], 0

    mov dl, [drive]
    mov ah, 42h
    int 13h
    jc .error

    popa
    pop ds
    popf
    clc
    ret

.error:
    mov al, '?'
    out 0xE9, al
    popa
    pop ds
    popf
    stc
    ret
